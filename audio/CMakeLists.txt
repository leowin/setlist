cmake_minimum_required(VERSION 3.22)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    project(setlistaudio CXX OBJCXX)
else()
    project(setlistaudio CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Platform-independent core: metronome, click synth, event sequencer.
add_library(audiocore STATIC src/Metronome.cpp src/Synth.cpp src/Listener.cpp)
target_include_directories(audiocore PUBLIC src)
target_compile_options(audiocore PRIVATE -O2 -ffast-math -fno-exceptions)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    # Static library for Kotlin/Native cinterop (shared/src/nativeInterop/cinterop/setlistaudio.def).
    add_library(setlistaudio_ios STATIC src/ios/CoreAudioEngine.mm src/ios/capi.cpp)
    target_include_directories(setlistaudio_ios PUBLIC src src/capi)
    target_link_libraries(setlistaudio_ios PUBLIC audiocore)
    target_compile_options(setlistaudio_ios PRIVATE -O2 -fobjc-arc)
elseif(ANDROID)
    find_package(oboe REQUIRED CONFIG)
    add_library(setlistaudio SHARED src/android/OboeEngine.cpp src/android/jni.cpp)
    target_link_libraries(setlistaudio PRIVATE audiocore oboe::oboe log)
    target_compile_options(setlistaudio PRIVATE -O2)
else()
    enable_testing()
    add_executable(transport_test tests/transport_test.cpp)
    target_link_libraries(transport_test PRIVATE audiocore)
    add_test(NAME transport_test COMMAND transport_test)
    add_executable(engine_test tests/engine_test.cpp)
    target_link_libraries(engine_test PRIVATE audiocore)
    add_test(NAME engine_test COMMAND engine_test ${CMAKE_CURRENT_SOURCE_DIR}/../shared/src/commonMain/composeResources/files/setlist.sf2)
    add_executable(listener_test tests/listener_test.cpp)
    target_link_libraries(listener_test PRIVATE audiocore)
    add_test(NAME listener_test COMMAND listener_test)
endif()
